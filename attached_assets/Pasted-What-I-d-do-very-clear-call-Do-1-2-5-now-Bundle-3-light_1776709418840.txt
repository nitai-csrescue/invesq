// services/ai/scoreSignals.ts
export function scoreSignals(ctx) {
  const { deployment, edges, nodes, resources } = ctx;

  const signals = [];

  // Deployment health
  if (deployment.healthScore < 70) {
    signals.push({ type: "risk", label: "Low deployment health", weight: 3, source: ["deployment", deployment.id] });
  }

  // Blockers
  deployment.blockers.forEach(b =>
    signals.push({ type: "risk", label: b.label, weight: b.severity === "high" ? 3 : 2, source: ["blocker", b.id] })
  );

  // Degraded edges
  edges.filter(e => e.status === "degraded").forEach(e =>
    signals.push({ type: "risk", label: `Degraded: ${e.label}`, weight: 2, source: ["edge", e.id] })
  );

  // Strong positives (e.g., adoption, BI usage)
  // (use your metrics to add “opportunity” signals)

  return signals.sort((a,b)=>b.weight-a.weight);
}
